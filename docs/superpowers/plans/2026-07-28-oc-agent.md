# OC Agent Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a single-file Lua agent that runs inside OpenComputers, connects to OpenAI-compatible LLM APIs, and provides tool-calling capabilities including autonomous Lua code execution.

**Architecture:** Single file `agent.lua` with internal sections for JSON codec, HTTP client, tool system, LLM client, config/history management, and REPL loop. Pi agent's two-level loop (outer REPL + inner tool-calling) adapted for OC's event-driven constraints.

**Tech Stack:** Pure Lua 5.2/5.3, OpenOS APIs (internet, filesystem, component, computer, term, event, serialization, shell), OpenAI-compatible chat completions API

## Global Constraints

- Single file: `agent.lua`, no external dependencies
- Must yield periodically to avoid OC timeout crashes
- Target: ~550 lines total
- API format: OpenAI /v1/chat/completions with native tool_calls
- Config format: `serialization.serialize` (Lua code format)
- Hardware: Internet Card T2 (HTTPS), RAM T2+, HDD T1+
- No collectgarbage, no binary load, no C libraries

```lua
local json = {}

function json.encode(val)
  local t = type(val)
  if val == nil then return "null" end
  if t == "boolean" then return val and "true" or "false" end
  if t == "number" then
    if val ~= val or val == math.huge or val == -math.huge then return "null" end
    return tostring(val)
  end
  if t == "string" then
    local s = val:gsub("[\\\"\n\r\t]", {["\\"]="\\\\", ['\"']='\\\"', ["\n"]="\\n", ["\r"]="\\r", ["\t"]="\\t"})
    return '"' .. sF .. '"'
  end
  if t == "table" then
    -- check if array-like: keys are 1..n with no gaps and no string keys
    local is_array = true
    local max_idx = 0
    for k, _ in pairs(val) do
      if type(k) ~= "number" or k < 1 or k ~= math.floor(k) then
        is_array = false
        break
      end
      if k > max_idx then max_idx = k end
    end
    if is_array and max_idx == #val then
      local parts = {}
      for i = 1, #val do parts[i] = json.encode(val[i]) end
      return "[" .. table.concat(parts, ",") .. "]"
    else
      local parts = {}
      for k, v in pairs(val) do
        if type(k) == "string" then
          parts[#parts + 1] = json.encode(k) .. ":" .. json.encode(v)
        end
      end
      return "{" .. table.concat(parts, ",") .. "}"
    end
  end
  error("json: cannot encode type " .. t)
end
```

```lua
function json.decode(str)
  local pos = 1
  local function skip_ws()
    pos = (str:find("^%s*", pos) or pos)
  end
  local function parse_value()
    skip_ws()
    local c = str:sub(pos, pos)
    if c == '"' then return parse_string()
    elseif c == '{' then return parse_object()
    elseif c == '[' then return parse_array()
    elseif c == 't' then pos = pos + 4; return true
    elseif c == 'f' then pos = pos + 5; return? false
    elseif c == 'n' then pos = pos + 4; return nil
    elseif c == '-' or (c >= '0' and c <= '9') then return parse_number()
    else error("json: unexpected char at " .. pos) end
  end
  -- parse_string: handle \" \\ \n \r \t \/
  -- parse_number: integer and float, including negative and scientific
  -- parse_array: [ value, value, ... ]
  -- parse_object: { "key": value, ... }
  -- (full implementations in actual code)
  local result = parse_value()
  skip_ws()
  if pos <= #str then error("json: trailing data at " .. pos) end
  return result
end
```

```lua
local function http_post(url, headers, body)
  local internet = require("internet")
  local ok, handle = pcall(function()
    return internet.request(url, body, headers, "POST")
  end)
  if not ok then return nil, nil, "connection failed: " .. tostring(handle) end

  local chunks = {}
  for chunk in handle do
    chunks[#chunks + 1] = chunk
  end
  local response_body = table.concat(chunks)

  local mt = getmetatable(handle)
  local code, msg, resp_headers
  if mt and mt.__index and mt.__index.response then
    code, msg, resp_headers = mt.__index.response()
  end

  return code or 0, response_body, nil
end
```

```lua
local function execute_tool(name, args_str)
  local args, err = json.decode(args_str)
  if not args then return "Error parsing arguments: " .. tostring(err) end

  if name == "read_file" then
    local ok, result = pcall(function()
      local f = io.open(args.path, "r")
      if not f then error("file not found: " .. args.path) end
      local c = f:read("*a")
      f:close()
      return c
    end)
    return ok and result or ("Error: " .. tostring(result))

  elseif name == "write_file" then
    local ok, result = pcall(function()
      local f = io.open(args.path, "w")
      if not f then error("cannot open: " .. args.path) end
      f:write(args.content)
      f:close()
      return "Written to " .. args.path
    end)
    return ok and result or ("Error: " .. tostring(result))

  elseif name == "list_directory" then
    local ok, result = pcall(function()
      local fs = require("filesystem")
      local parts = {}
      for f in fs.list(args.path or "/") do parts[#parts+1] = f end
      return table.concat(parts, "\n")
    end)
    return ok and (result ~= "" and result or "(empty)") or ("Error: " .. tostring(result))

  elseif name == "execute_lua" then
    return execute_lua_code(args.code)

  elseif name == "component_list" then
    local ok, result = pcall(function()
      local comp = require("component")
      local parts = {}
      for addr, typ in comp.list(args.filter or "") do
        parts[#parts+1] = addr:sub(1,8) .. "... = " .. typ
      end
      return table.concat(parts, "\n")
    end)
    return ok and result or ("Error: " .. tostring(result))

  elseif name == "shell_execute" then
    local ok, result = pcall(function()
      local sh = require("shell")
      return sh.execute(args.command)
    end)
    return ok9 and tostring(result) or ("Error: " .. tostring(result))

  else
    return "Unknown tool: " .. name
  end
end
```

```lua
local function execute_lua_code(code)
  local original_write = io.write
  local captured = {}
  io.write = function(s)
    captured[#captured + 1] = tostring(s or "")
    return true
  end

  local fn, compile_err = load(code)
  local ok, result
  if fn then
    ok, result = pcall(fn)
  end

  io.write = original_write

  local output = table.concat(captured)
  if not fn then
    return "Compile error: " .. tostring(compile_err)
  elseif not ok then
    return "Runtime error: " .. tostring(result) .. (output ~= "" and ("\nOutput before error:\n" .. output) or "")
  else
    local ret = output
    if result ~= nil then ret = ret .. (ret ~= "" and "\n" or "") .. "=> " .. tostring(result) end
    return ret ~= "" and ret or "(no output)"
  end
end
```

- [ ] **Step 1: Write chat function**

```lua
local function chat(messages, config)
  local system_prompt = build_system_prompt()

  local api_messages = {}
  api_messages[#api_messages + 1] = {role="system", content=system_prompt}
  for _, msg in ipairs(messages) do
    api_messages[#api_messages + 1] = msg
  end

  local body = json.encode({
    model = config.model,
    messages = api_messages,
    tools = TOOLS,
    max_tokens = 2048,
    temperature = 0.7
  })

  local headers = {
    ["Content-Type"] = "application/json",
    ["Authorization"] = "Bearer " .. config.api_key
  }

  local code, resp, err = http_post(config.api_url, headers, body)
  if err then return {content = nil, tool_calls = nil, finish_reason = "error", error = err} end
  if not code or code ~= 200 then
    return {content = nil, tool_calls = nil, finish_reason = "error", error = "HTTP " .. tostring(code) .. ": " .. tostring(resp):sub(1, 200)}
  end

  local data, decode_err = json.decode(resp)
  if not data then
    return {content = nil, tool_calls = nil, finish_reason = "error", error = "JSON decode: " .. tostring(decode_err)}
  end

  local choice = data.choices and data.choices[1]
  if not choice then
    return {content = nil, tool_calls = nil, finish_reason = "error", error = "No choices in response"}
  end

  local msg = choice.message or {}
  return {
    content = msg.content,
    tool_calls = msg.tool_calls,
    finish_reason = choice.finish_reason
  }
end
```

```lua
local CONFIG_PATH = "/home/agent_config.txt"
local HISTORY_PATH = "/home/agent_history.txt"

local function load_config()
  local fs = require("filesystem")
  if not fs.exists(CONFIG_PATH) then return nil end
  local f = io.open(CONFIG_PATH, "r")
  if not f then return nil end
  local content = f:read("*a")
  f:close()
  local ser = require("serialization")
  local ok, data = pcall(ser.unserialize, content)
  if ok and type(data) == "table" then return data end
  return nil
end

local function save_config(config)
  local ser = require("serialization")
  local f = io.open(CONFIG_PATH, "w")
  f:write(ser.serialize(config))
  f:close()
end

local function first_run_setup()
  print("OC Agent - First Run Setup")
```

```lua
local MAX_HISTORY = 20

local function load_history()
  local fs = require("filesystem")
  if not fs.exists(HISTORY_PATH) then return {} end
  local f = io.open(HISTORY_PATH, "r")
  if not f then return {} end
  local content = f:read("*a")
  f:close()
  local ser = require("serialization")
  local ok, data = pcall(ser.unserialize, content)
  if ok and type(data) == "table" then return data end
  return {}
end

local function save_history(messages)
  local ser = require("serialization")
  local f = io.open(HISTORY_PATH, "w")
  f:write(ser.serialize(messages))
  f:close()
end

local function trim_history(messages)
  if #messages <= MAX_HISTORY then return messages end
  local trimmed = {}
  for i = #messages - MAX_HISTORY + 1, #messages do
    trimmed[#trimmed + 1] = messages[i]
  end
  return trimmed
end
```

```lua
local function handle_command(cmd, config, messages)
  local parts = {}
  for w in cmd:gmatch("%S+") do parts[#parts+1] = w end
  local command = parts[1]

  if command == "/model" then
    config.model = parts[2] or config.model
    save_config(config)
    print("Model: " .. config.model)
  elseif command == "/key" then
    config.api_key = parts[2] or config.api_key
    save_config(config)
    print("API key updated")
  elseif command == "/url" then
    config.api_url = parts[2] or config.api_url
    save_config(config)
    print("API URL: " .. config.api_url)
  elseif command == "/reset" then
    messages = {}
    save_history(messages)
    print("History cleared")
  elseif command == "/hist" then
    print(#messages .. " messages in history")
  elseif command == "/tools" then
    for _, t in ipairs(TOOLS) do print("  " .. t.function.name .. ": " .. t.function.description) end
  elseif command == "/help" then
    print("Commands: /model /key /url /reset /hist /tools /help /exit")
  elseif command == "/exit" then
    return true, config, messages
  else
    print("Unknown command: " .. command .. ". Type /help for commands.")
  end
  return false, config, messages
end
```

- [ ] **Step 2: Write main REPL loop**

```lua
local function main()
  local component = require("component")
  if not component.isAvailable("internet") then
    print("Error: No internet card found. Tier 2 Internet Card required.")
    return
  end

  local config = load_config()
  if not config then config = first_run_setup() end

  local messages = load_history()
  local history = {}

  print("OC Agent ready. Model: " .. config.model .. ". Type /help for commands.")

  while true do
    io.write("> ")
    local input = io.read()
    if not input then break end
    input = input:gsub("\n", "")
    if input == "" then goto continue end

    if input:sub(1, 1) == "/" then
      local exit, c, m = handle_command(input, config, messages)
      config, messages = c, m
      if exit then break end
      goto continue
    end

    messages[#messages + 1] = {role = "user", content = input}
    history[#history + 1] = input

    while true do
      io.write("Thinking...\r")
      local response = chat(messages, config)

      if response.error then
        print("Error: " .. response.error)
        break
      end

      if response.content then
        print(response.content)
      end

      local assistant_msg = {role = "assistant", content = response.content or ""}
      if response.tool_calls then
        assistant_msg.tool_calls = response.tool_calls
      end
      messages[#messages + 1] = assistant_msg

      if not response.tool_calls or #response.tool_calls == 0 then
        break
      end

      for _, tc in ipairs(response.tool_calls) do
        local tool_name = tc.function.name
        local tool_args = tc.function.arguments
        print("[tool] " .. tool_name)
        local result = execute_tool(tool_name, tool_args)
        messages[#messages + 1] = {
          role = "tool",
          tool_call_id = tc.id,
          content = result
        }
      end
    end

    messages = trim_history(messages)
    save_history(messages)

    ::continue::
  end

  print("Goodbye!")
end
```
